Description: AWS CloudFormation Template that provisions a DMS Replication Instance (CloudWatch Alarms)
################################## Metadata ###############################################
Metadata: 
  AWS::CloudFormation::Interface: 
    ParameterGroups: 
      - 
        Label: 
          default: "Project Details"
        Parameters: 
          - AppCode
          - AppCodeLower
          - Environment
          - BU
          - BULower
          - Purpose
          - Requestor
          - OwnerContact
          - SupportContact
      - 
        Label: 
          default: "DMS Replication Instance Details"
        Parameters: 
          - ReplicationInstanceName
          - ReplicationInstanceType
          - ReplicationInstanceAZ
          - AllocatedStorage
          - SecurityGroups
          - VPCShortName
          - AutoMinorVersionUpgrade
          - PreferredMaintenanceWindow
          - WeekendOperation
          - AutomateShutdown
      - 
        Label: 
          default: "CloudWatch Alarms Details"
        Parameters: 
          - EmailAddress
          - ThresholdValue10PCT
          - ThresholdValue20PCT
          - ThresholdValue30PCT

################################## Parameters ###############################################
Parameters:
  
  AppCode:
    Type: String
    MaxLength: 5
    AllowedPattern: '^([A-Z0-9_-]){5}$'
    Description: The application code name (NEECode) cannot be of more that 5 characters
    ConstraintDescription: Appcode field does not match the required constraints
    Default: EN411

  AppCodeLower:
    Type: String
    MaxLength: 5
    AllowedPattern: '^([a-z0-9_-]){5}$'
    Description: The application code name (NEECode) cannot be of more that 5 characters
    ConstraintDescription: Appcode field does not match the required constraints
    Default: en411

  Environment:
    Type: String
    AllowedValues:
      - Prod
      - Test
      - Dev
      - QA
      - DR
      - Sandbox
    Default: Dev

  BU:
    Type: String
    Description: What is the Business Unit?
    AllowedValues:
      - NUC
      - PD
      - PGD
      - CS
      - IT
      - AllBU
      - GASINFRA

  BULower:
    Type: String
    Description: What is the Business Unit?
    AllowedValues:
      - nuc
      - pd
      - pgd
      - cs
      - it
      - allbu 
      - gasinfra

  Purpose:
    Type: String
    Description: Enter a brief summary of the replication instance purpose

  ReplicationInstanceName:
    Type: String
    MinLength: '1'
    MaxLength: '50'
    #AllowedPattern: '[a-zA-Z][a-zA-Z0-9]*'
    Description: Must be unique and 1-50 characters long. Must begin with a letter and contain only alphanumeric characters, dashes and underscores
    ConstraintDescription: 'Must be unique and 1-50 characters long. Must begin with a letter and contain only alphanumeric characters, dashes and underscores.'

  ReplicationInstanceType:
    Description: Choose the type of instance that you want to use for Replication Instance
    Type: String
    AllowedValues:
      - dms.t2.micro
      - dms.t2.small
      - dms.t2.medium
      - dms.t2.large
      - dms.c4.large
      - dms.c4.xlarge
      - dms.c4.2xlarge
      - dms.c4.4xlarge
    Default: dms.t2.medium

  ReplicationInstanceAZ:
    Description: Define what AvailabilityZone the EBS volume will be created in
    Type: String
    Default: ""
    AllowedValues:
      - ""
      - us-east-1a
      - us-east-1b
      - us-east-1c
      - us-east-1d
      - us-east-1e
      - us-east-1f

  AllocatedStorage:
    Description: Define the size in GB for the volume that you want to create
    Type: String
    Default: 30
    AllowedPattern: "[0-9]*"

  Requestor:
    Type: String
    Description: Please specify email address of the requestor

  OwnerContact:
      Type: String
      Description: Please specify email address of the owner
      AllowedPattern: '^([a-zA-Z0-9_\-\.]+)@([a-zA-Z0-9_\-\.]+)\.([a-zA-Z]{2,5})$'
      ConstraintDescription: OwnerContact field must be an email address
      Default: Nelson.Castellanos@fpl.com
      
  SupportContact:
      Type: String
      Description: Please specify email address of the support contact
      AllowedPattern: '^([a-zA-Z0-9_\-\.]+)@([a-zA-Z0-9_\-\.]+)\.([a-zA-Z]{2,5})$'
      ConstraintDescription: SupportContact field must be an email address
      Default: DL-EDP-PLATFORM@nexteraenergy.com

  SecurityGroups:
    Type: List<AWS::EC2::SecurityGroup::Id>
    Description: Please specify security groups to be attached to the Replication Instance

  VPCShortName:
    Type: 'AWS::SSM::Parameter::Value<String>'
    Description: 'Replication Instance VPC ID'
    Default: '/account/VPCShortName'

  AutoMinorVersionUpgrade:
    Type: String
    Description: Allow minor engine upgrades to be automatically made to the instance during maintenance window
    AllowedValues:
      - 'true'
      - 'false'
    Default: 'true'

  PreferredMaintenanceWindow: 
    Type: String
    Description: Please specify a preferred weekly maintenance window (30 mins in UTC)
    Default: Sun:19:00-Sun:23:00

  WeekendOperation:
    Type: String
    Description: Please specify if the replication instance conducts weekend operations
    AllowedValues:
      - 'true'
      - 'false'

  AutomateShutdown:
    Type: String
    Description: This field specifies if the server needs to be automatically shut down every day
    AllowedValues:
      - "yes"
      - "no"
    Default: "no"

  EmailAddress: 
    Description: Please provide the Email Name or Distribution list Name
    Type: String
    AllowedValues:
      - DL-EDP-PLATFORM
      - DL-EDP-SUPPORT
    Default: DL-EDP-SUPPORT
  
  ThresholdValue30PCT:
    Description: Please provide the 30% of the "AllocatedStorage" expressed in exponential (1e9 = 1GB)
    Type: Number
  
  ThresholdValue20PCT:
    Description: Please provide the 20% of the "AllocatedStorage" expressed in exponential (1e9 = 1GB)
    Type: Number
  
  ThresholdValue10PCT:
    Description: Please provide the 10% of the "AllocatedStorage" expressed in exponential (1e9 = 1GB)
    Type: Number

################################## Mappings ##################################################
Mappings:
  RegionMap:
    us-east-1:
      Region: V
      RegionLower: v

################################## Conditions ##################################################
Conditions: 
  IsNotProd: !Not [!Equals [!Ref Environment, 'Prod']]

####################################### Resources #############################################

Resources:
  ReplicationInstance:
    Type: "AWS::DMS::ReplicationInstance"
    Properties:
      AllocatedStorage: !Ref AllocatedStorage
      AutoMinorVersionUpgrade: !Ref AutoMinorVersionUpgrade
      AvailabilityZone: !Ref ReplicationInstanceAZ
      KmsKeyId: !ImportValue Platform-Default-KMS
      MultiAZ: 'false'
      PreferredMaintenanceWindow: !Ref PreferredMaintenanceWindow
      PubliclyAccessible: false
      ReplicationInstanceClass: !Ref ReplicationInstanceType
      ReplicationSubnetGroupIdentifier: !ImportValue Platform-Default-DMS-Subnet-Group
      ReplicationInstanceIdentifier: !Join [ "-", [ 'dms', !Join ["", [ !ImportValue Account-Global-AccountCodeLower, !FindInMap [RegionMap, !Ref "AWS::Region", RegionLower]]], !Ref AppCodeLower, !Ref BULower, !Ref ReplicationInstanceName ]]
      Tags:
        - Key: AppCode
          Value: !Ref AppCode
        - Key: Environment
          Value: !Ref Environment
        - Key: Requestor
          Value: !Ref Requestor
        - Key: BU
          Value: !Ref BU
        - Key: Purpose
          Value: !Ref Purpose
        - Key: OwnerContact
          Value: !Ref OwnerContact
        - Key: SupportContact
          Value: !Ref SupportContact
        - Key: AutomateShutdown
          Value: !Ref AutomateShutdown
        - Key: WeekendOperation
          Value: !Ref WeekendOperation
        - Key: StackName
          Value: !Ref 'AWS::StackName'

      VpcSecurityGroupIds: !Ref SecurityGroups

    Type: "AWS::CloudWatch::Alarm"
    Properties:
      Namespace: AWS/DMS

            - " "
            - "environment="
            - !Ref Environment 
            - " "
            - "emailAddress="
            - Ref: EmailAddress
      AlarmName:
        Fn::Join:
          - "_"
          - - "en411_edp_cwa_cpu_utilization_70"
            - Ref: ReplicationInstanceName             
      ComparisonOperator: "GreaterThanThreshold"
      Dimensions:
        - Name: ReplicationInstanceIdentifier
          Value: !Join [ "-", [ 'dms', !Join ["", [ !ImportValue Account-Global-AccountCodeLower, !FindInMap [RegionMap, !Ref "AWS::Region", RegionLower]]], !Ref AppCodeLower, !Ref BULower, !Ref ReplicationInstanceName ]]
      MetricName: "CPUUtilization"
      Namespace: "AWS/DMS"
      Period: 300 
      EvaluationPeriods: 1
      Statistic: Average
      Threshold: 70
      TreatMissingData: "missing"
